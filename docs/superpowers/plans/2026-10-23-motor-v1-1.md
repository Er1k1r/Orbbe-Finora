## Global Constraints

- Valores monetários são **inteiros em centavos**; taxas são frações (0.01 = 1% a.m.); meses `'YYYY-MM'`; datas `'YYYY-MM-DD'`.
- `src/domain` não importa React, Dexie, `ui/` nem `data/`. `src/data` importa do domínio só tipos (e o seed pode importar tipos).
- **Movimento interno** = lançamento com `especie` (`'transferencia' | 'ajuste'`): entra no saldo das contas e em nada mais (receita, despesa, sobra, pendentes do resumo, renda, custo essencial, insights, projeção).
- **Pagamento de financiamento** = saída com `financiamentoId` de financiamento existente (`ehPagamentoParcela`): nunca é despesa comum; a parcela do mês vem do contrato.
- Toda tarefa mantém **suíte completa, `npm run build` e `npm run lint` verdes** (rodar de `app/`). Nunca editar um valor esperado sem refazer a conta à mão e registrar no relatório.
- Cada comportamento novo precisa de um teste que falhe se o comportamento for removido.
- Sem migração de dados: o app não foi publicado; banco e backup sobem para a versão 2.

**User decisions (already made):** pagar parcela reduz o saldo devedor, mostrando total, juros e desconto de quitação/antecipação · cadastro do financiamento pelo carnê com taxa calculada · cheque especial automático (saldo negativo permitido; se o banco não está negativo, falta lançamento) · custo essencial inclui lançados e futuros (ex.: óculos R$ 50 × 12, parcela 4/12) · incluir: comprometido nos próximos meses, transferência entre contas, conta reserva + fim de recorrência, descrição + armazenamento persistente.

```ts
/** Soma n meses a uma data 'YYYY-MM-DD', mantendo o dia (limitado ao último dia do mês de destino). */
export function somarMeses(data: string, n: number): string {
  const [y, m, d] = data.split('-').map(Number)
  const alvo = new Date(Date.UTC(y, m - 1 + n, 1)).toISOString().slice(0, 7)
  const ultimo = Number(fimDoMes(alvo).slice(8, 10))
  return `${alvo}-${String(Math.min(d, ultimo)).padStart(2, '0')}`
}

/** Quantos meses de `de` até `ate` ('YYYY-MM'); negativo se `ate` vier antes. */
export function diffMeses(de: Mes, ate: Mes): number {
  const [y1, m1] = de.split('-').map(Number)
  const [y2, m2] = ate.split('-').map(Number)
  return (y2 - y1) * 12 + (m2 - m1)
}
```

```ts
import { gerarCronograma, type Parcela } from './amortizacao'
import { diffMeses, mesDe, somarMeses } from './meses'
import type { Centavos, Contrato, Mes, PagamentoFinanciamento, StatusLancamento } from './tipos'

export type DadosCarne = Pick<Contrato, 'valorFinanciado' | 'valorParcela' | 'totalParcelas' | 'sistema'>

const pmtExato = (pv: number, i: number, n: number) => (i === 0 ? pv / n : (pv * i) / (1 - Math.pow(1 + i, -n)))

/** Taxa mensal implícita no carnê; null quando os dados não fecham (parcela pequena demais). */
export function taxaImplicita(c: DadosCarne): number | null {
  const { valorFinanciado: pv, valorParcela: p, totalParcelas: n } = c
  if (!(pv > 0) || !(p > 0) || !Number.isInteger(n) || n < 1) return null
  if (c.sistema === 'SAC') {
    const i = (p - pv / n) / pv
    return i >= 0 ? i : null
  }
  if (p * n < pv) return null
  if (p * n === pv) return 0
  let lo = 0
  let hi = 1
  for (let k = 0; k < 200; k++) {
    const mid = (lo + hi) / 2
    if (pmtExato(pv, mid, n) > p) hi = mid
    else lo = mid
  }
  return (lo + hi) / 2
}

export interface PagamentoContrato { data: string; valor: Centavos; tipo: PagamentoFinanciamento; status: StatusLancamento }
export interface ParcelaContrato extends Parcela { vencimento: string }
export interface EstadoContrato {
  cronogramaOriginal: ParcelaContrato[]
  parcelasPagas: number
  saldoDevedor: Centavos
  jurosPagos: Centavos
  antecipado: Centavos
  restantes: ParcelaContrato[] // `numero` = posição no contrato
  jurosAPagar: Centavos
  totalContrato: Centavos
  restanteNominal: Centavos
  descontoQuitacao: Centavos
}

const vencimentoDe = (c: Contrato, numero: number) => somarMeses(c.primeiraParcela, numero - 1)

function cronograma(c: Contrato, saldo: Centavos, prazo: number): Parcela[] {
  return gerarCronograma({
    saldo, taxa: c.taxaMensal, prazo, sistema: c.sistema,
    parcelaFixa: c.sistema === 'PRICE' ? c.valorParcela : undefined,
    amortizacaoFixa: c.sistema === 'SAC' ? Math.round(c.valorFinanciado / c.totalParcelas) : undefined,
  })
}

/** Estado do contrato após as parcelas pagas antes do app e os pagamentos pagos (até `ate`, se dado). */
export function estadoContrato(c: Contrato, pagamentos: PagamentoContrato[], ate?: string): EstadoContrato {
  const cronogramaOriginal = cronograma(c, c.valorFinanciado, c.totalParcelas)
    .map((p) => ({ ...p, vencimento: vencimentoDe(c, p.numero) }))
  let saldo = c.valorFinanciado
  let pagas = 0
  let jurosPagos = 0
  let antecipado = 0
  const passoRegular = () => {
    const p = cronograma(c, saldo, c.totalParcelas - pagas)[0]
    if (!p) return
    jurosPagos += p.juros
    saldo = p.saldo
    pagas += 1
  }
  for (let k = 0; k < Math.min(c.parcelasPagasAntes, c.totalParcelas); k++) passoRegular()
  const validos = pagamentos
    .filter((p) => p.status === 'pago' && (ate === undefined || p.data <= ate))
    .sort((a, b) => a.data.localeCompare(b.data))
  for (const p of validos) {
    if (saldo <= 0) break
    if (p.tipo === 'antecipacao') {
      const v = Math.min(p.valor, saldo)
      saldo -= v
      antecipado += v
    } else passoRegular()
  }
  const restantes = cronograma(c, saldo, c.totalParcelas - pagas).map((p, i) => ({
    ...p, numero: pagas + i + 1, vencimento: vencimentoDe(c, pagas + i + 1),
  }))
  const soma = (xs: Parcela[], f: (p: Parcela) => number) => xs.reduce((s, p) => s + f(p), 0)
  const restanteNominal = soma(restantes, (p) => p.parcela)
  return {
    cronogramaOriginal, parcelasPagas: pagas, saldoDevedor: saldo, jurosPagos, antecipado, restantes,
    jurosAPagar: soma(restantes, (p) => p.juros),
    totalContrato: soma(cronogramaOriginal, (p) => p.parcela),
    restanteNominal,
    descontoQuitacao: restanteNominal - saldo,
  }
}

/** Valor presente (à taxa do contrato) das últimas `n` parcelas restantes e o desconto em relação ao nominal. */
export function antecipacaoUltimas(e: EstadoContrato, taxa: number, n: number) {
  const ult = e.restantes.slice(-Math.max(0, n))
  const offset = e.restantes.length - ult.length
  const valor = Math.round(ult.reduce((s, p, k) => s + p.parcela / Math.pow(1 + taxa, offset + k + 1), 0))
  const nominal = ult.reduce((s, p) => s + p.parcela, 0)
  return { valor, nominal, desconto: nominal - valor }
}

/** Parcela que vence em `mes`: já paga → valor do cronograma original; em aberto → restante; fora do contrato → 0. */
export function parcelaNoMes(c: Contrato, e: EstadoContrato, mes: Mes): Centavos {
  const k = diffMeses(mesDe(c.primeiraParcela), mes) + 1
  if (k < 1 || k > c.totalParcelas) return 0
  if (k <= e.parcelasPagas) return e.cronogramaOriginal[k - 1]?.parcela ?? 0
  return e.restantes.find((p) => p.numero === k)?.parcela ?? 0
}

/** Soma das parcelas em aberto que venceram antes de `mes`. */
export function parcelasAtrasadas(e: EstadoContrato, mes: Mes): Centavos {
  return e.restantes.filter((p) => mesDe(p.vencimento) < mes).reduce((s, p) => s + p.parcela, 0)
}
```

```ts
// parcelamento.ts
import { somarMeses } from './meses'
import type { Centavos, Lancamento, StatusLancamento } from './tipos'

export interface NovoParcelamento {
  grupoId: string
  valorParcela: Centavos
  total: number
  parcelaAtual: number
  dataParcelaAtual: string
  statusAtual: StatusLancamento
  base: Omit<Lancamento, 'id' | 'valor' | 'data' | 'status' | 'parcelamento' | 'recorrencia' | 'origemId' | 'geradoAte' | 'recorrenciaAte'>
}

export function criarParcelamento(e: NovoParcelamento): Lancamento[] {
  if (!Number.isInteger(e.total) || e.total < 2) throw new Error('Uma compra parcelada precisa de pelo menos 2 parcelas.')
  if (!Number.isInteger(e.parcelaAtual) || e.parcelaAtual < 1 || e.parcelaAtual > e.total)
    throw new Error('A parcela atual deve estar entre 1 e o total de parcelas.')
  if (!Number.isSafeInteger(e.valorParcela) || e.valorParcela <= 0) throw new Error('O valor da parcela deve ser maior que zero.')
  const out: Lancamento[] = []
  for (let n = e.parcelaAtual; n <= e.total; n++) {
    out.push({
      ...e.base,
      id: `${e.grupoId}:${n}`,
      valor: e.valorParcela,
      data: somarMeses(e.dataParcelaAtual, n - e.parcelaAtual),
      status: n === e.parcelaAtual ? e.statusAtual : 'pendente',
      parcelamento: { grupoId: e.grupoId, numero: n, total: e.total },
    })
  }
  return out
}

export function resumoParcelamento(grupo: Lancamento[]) {
  const ordenado = [...grupo].sort((a, b) => (a.parcelamento?.numero ?? 0) - (b.parcelamento?.numero ?? 0))
  const pend = ordenado.filter((l) => l.status === 'pendente')
  return {
    pagas: ordenado.length - pend.length,
    restantes: pend.length,
    valorRestante: pend.reduce((s, l) => s + l.valor, 0),
    ultimaData: ordenado[ordenado.length - 1]?.data,
    total: ordenado[0]?.parcelamento?.total ?? 0,
  }
}
```

```ts
// armazenamento.ts
export type EstadoArmazenamento = 'persistente' | 'nao_persistente' | 'indisponivel'
interface NavLike { storage?: { persist?: () => Promise<boolean>; persisted?: () => Promise<boolean> } }

export async function pedirPersistencia(nav: NavLike | undefined = globalThis.navigator as NavLike | undefined): Promise<EstadoArmazenamento> {
  const s = nav?.storage
  if (!s?.persist || !s.persisted) return 'indisponivel'
  if (await s.persisted()) return 'persistente'
  return (await s.persist()) ? 'persistente' : 'nao_persistente'
}
```
